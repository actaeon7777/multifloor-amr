cmake_minimum_required(VERSION 3.8)
project(amr_depth_filter)
# The workspace builds without a build type (-O0); this per-point loop runs at
# 10 Hz on the Jetson, where -O0 cost ~18 % of a core.
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(depth_persistence_filter src/depth_persistence_filter.cpp)
target_compile_features(depth_persistence_filter PUBLIC cxx_std_17)
target_include_directories(depth_persistence_filter PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
ament_target_dependencies(depth_persistence_filter rclcpp sensor_msgs tf2 tf2_ros)
install(TARGETS depth_persistence_filter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_persistence test/test_persistence.cpp)
  target_compile_features(test_persistence PUBLIC cxx_std_17)
  target_include_directories(test_persistence PUBLIC include)
endif()
ament_package()
